83template<typename Float_>
84FindMedianMadResults<Float_> find_median_mad(std::size_t num_obs, Float_* metrics, const FindMedianMadOptions& options) {
85 static_assert(std::is_floating_point<Float_>::value);
86
87 // Rotate all the NaNs to the front of the buffer and ignore them.
88 I<decltype(num_obs)> lost = 0;
89 for (I<decltype(num_obs)> i = 0; i < num_obs; ++i) {
90 if (std::isnan(metrics[i])) {
91 std::swap(metrics[i], metrics[lost]);
92 ++lost;
93 }
94 }
95 metrics += lost;
96 num_obs -= lost;
97
98 if (options.log) {
99 for (I<decltype(num_obs)> i = 0; i < num_obs; ++i) {
100 auto& val = metrics[i];
101 if (val > 0) {
102 val = std::log(val);
103 } else if (val == 0) {
104 val = -std::numeric_limits<double>::infinity();
105 } else {
106 throw std::runtime_error("cannot log-transform negative values");
107 }
108 }
109 }
110
111 const auto median = quickstats::median<Float_>(num_obs, metrics);
112 if (options.median_only || std::isnan(median)) {
113 // Giving up.
114 return FindMedianMadResults<Float_>(median, std::numeric_limits<Float_>::quiet_NaN());
115 } else if (std::isinf(median)) {
116 // MADs should be no-ops when added/subtracted from infinity. Any
117 // finite value will do here, so might as well keep it simple.
118 return FindMedianMadResults<Float_>(median, static_cast<Float_>(0));
119 }
120
121 // As an aside, there's no way to avoid passing in 'metrics' as a Float_,
122 // even if the original values were integers, because we need to do this
123 // subtraction here that could cast integers to floats. So at some point we
124 // will need a floating-point buffer, and so we might as well just pass the
125 // metrics in as floats in the first place. Technically the first sort
126 // could be done with an integer buffer but then we'd need an extra argument.
127
128 for (I<decltype(num_obs)> i = 0; i < num_obs; ++i) {
129 metrics[i] = std::abs(metrics[i] - median);
130 }
131 auto mad = quickstats::median<Float_>(num_obs, metrics);
132 mad *= 1.4826; // for equivalence with the standard deviation under normality.
133
134 return FindMedianMadResults<Float_>(median, mad);
135}
