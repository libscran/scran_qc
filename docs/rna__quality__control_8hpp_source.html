93void compute_rna_qc_metrics(
94 const tatami::Matrix<Value_, Index_>& mat,
95 const std::vector<Subset_>& subsets,
96 const ComputeRnaQcMetricsBuffers<Sum_, Detected_, Proportion_>& output,
97 const ComputeRnaQcMetricsOptions& options)
98{
99 auto NC = mat.ncol();
100 auto nsubsets = subsets.size();
101
102 PerCellQcMetricsBuffers<Sum_, Detected_, Value_, Index_> tmp;
103 tmp.sum = output.sum;
104 tmp.detected = output.detected;
105
106 constexpr bool same_type = std::is_same<Sum_, Proportion_>::value;
107 typename std::conditional<same_type, bool, std::vector<std::vector<Sum_> > >::type placeholder_subset;
108
109 if (output.subset_proportion.size()) {
110 // Providing space for the subset sums if they're not the same type.
111 if constexpr(same_type) {
112 tmp.subset_sum = output.subset_proportion;
113 } else {
114 placeholder_subset.resize(sanisizer::cast<decltype(placeholder_subset.size())>(nsubsets));
115 tmp.subset_sum.resize(sanisizer::cast<decltype(tmp.subset_sum.size())>(nsubsets));
116 for (decltype(nsubsets) s = 0; s < nsubsets; ++s) {
117 auto& b = placeholder_subset[s];
118 tatami::resize_container_to_Index_size(b, NC);
119 tmp.subset_sum[s] = b.data();
120 }
121 }
122 }
123
124 PerCellQcMetricsOptions opt;
125 opt.num_threads = options.num_threads;
126 per_cell_qc_metrics(mat, subsets, tmp, opt);
127
128 for (decltype(nsubsets) s = 0 ; s < nsubsets; ++s) {
129 auto dest = output.subset_proportion[s];
130 if (dest) {
131 auto src = tmp.subset_sum[s];
132 for (Index_ c = 0; c < NC; ++c) {
133 dest[c] = static_cast<Proportion_>(src[c]) / static_cast<Proportion_>(tmp.sum[c]);
134 }
135 }

284 }

307template<class Host_, typename Sum_, typename Detected_, typename Proportion_, typename BlockSource_, typename Output_>
308void rna_filter(const Host_& host, std::size_t n, const ComputeRnaQcMetricsBuffers<Sum_, Detected_, Proportion_>& metrics, BlockSource_ block, Output_* output) {
309 constexpr bool unblocked = std::is_same<BlockSource_, bool>::value;
310 std::fill_n(output, n, 1);
311
312 const auto& sum = host.get_sum();
313 for (decltype(n) i = 0; i < n; ++i) {
314 auto thresh = [&]{
315 if constexpr(unblocked) {
316 return sum;
317 } else {
318 return sum[block[i]];
319 }
320 }();
321 output[i] = output[i] && (metrics.sum[i] >= thresh);
322 }
323
324 const auto& detected = host.get_detected();
325 for (decltype(n) i = 0; i < n; ++i) {
326 auto thresh = [&]{
327 if constexpr(unblocked) {
328 return detected;
329 } else {
330 return detected[block[i]];
331 }
332 }();
333 output[i] = output[i] && (metrics.detected[i] >= thresh);
334 }
335
336 auto nsubsets = metrics.subset_proportion.size();
337 for (decltype(nsubsets) s = 0; s < nsubsets; ++s) {
338 auto sub = metrics.subset_proportion[s];
339 const auto& sthresh = host.get_subset_proportion()[s];
340 for (decltype(n) i = 0; i < n; ++i) {
341 auto thresh = [&]{
342 if constexpr(unblocked) {
343 return sthresh;
344 } else {
345 return sthresh[block[i]];
346 }
347 }();
348 output[i] = output[i] && (sub[i] <= thresh);
349 }
350 }
351}
