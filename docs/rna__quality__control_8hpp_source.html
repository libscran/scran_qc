104void compute_rna_qc_metrics(
105 const tatami::Matrix<Value_, Index_>& mat,
106 const std::vector<Subset_>& subsets,
107 const ComputeRnaQcMetricsBuffers<Sum_, Detected_, Proportion_>& output,
108 const ComputeRnaQcMetricsOptions& options)
109{
110 const auto NC = mat.ncol();
111 const auto nsubsets = subsets.size();
112
113 PerCellQcMetricsBuffers<Sum_, Detected_, Value_, Index_> tmp;
114 tmp.sum = output.sum;
115 tmp.detected = output.detected;
116
117 constexpr bool same_type = std::is_same<Sum_, Proportion_>::value;
118 typename std::conditional<same_type, bool, std::vector<std::vector<Sum_> > >::type placeholder_subset;
119
120 if (output.subset_proportion.size()) {
121 // Providing space for the subset sums if they're not the same type.
122 if constexpr(same_type) {
123 tmp.subset_sum = output.subset_proportion;
124 } else {
125 sanisizer::resize(placeholder_subset, nsubsets);

127 for (I<decltype(nsubsets)> s = 0; s < nsubsets; ++s) {
128 auto& b = placeholder_subset[s];
129 tatami::resize_container_to_Index_size(b, NC);
130 tmp.subset_sum[s] = b.data();
131 }
132 }
133 }
134
135 PerCellQcMetricsOptions opt;
136 opt.num_threads = options.num_threads;
137 opt.subset_containers_have_indices = options.subset_containers_have_indices;
138 per_cell_qc_metrics(mat, subsets, tmp, opt);
139
140 for (I<decltype(nsubsets)> s = 0 ; s < nsubsets; ++s) {
141 const auto dest = output.subset_proportion[s];
142 if (dest) {
143 const auto src = tmp.subset_sum[s];
144 for (Index_ c = 0; c < NC; ++c) {

337template<class Host_, typename Sum_, typename Detected_, typename Proportion_, typename BlockSource_, typename Output_>
338void rna_filter(
339 const Host_& host,
340 const std::size_t num_cells,
341 const ComputeRnaQcMetricsBuffers<Sum_, Detected_, Proportion_>& metrics,
342 BlockSource_ block,
343 Output_* const output
344) {
345 constexpr bool unblocked = std::is_same<BlockSource_, bool>::value;
346 std::fill_n(output, num_cells, 1);
347
348 const auto& sum = host.get_sum();
349 for (I<decltype(num_cells)> i = 0; i < num_cells; ++i) {
350 auto thresh = [&]{
351 if constexpr(unblocked) {
352 return sum;
353 } else {
354 return sum[block[i]];
355 }
356 }();
357 output[i] = output[i] && (metrics.sum[i] >= thresh);
358 }
359
360 const auto& detected = host.get_detected();
361 for (I<decltype(num_cells)> i = 0; i < num_cells; ++i) {
362 auto thresh = [&]{
363 if constexpr(unblocked) {
364 return detected;
365 } else {
366 return detected[block[i]];
367 }
368 }();
369 output[i] = output[i] && (metrics.detected[i] >= thresh);
370 }
371
372 const auto nsubsets = metrics.subset_proportion.size();
373 for (I<decltype(nsubsets)> s = 0; s < nsubsets; ++s) {
374 const auto sub = metrics.subset_proportion[s];
375 const auto& sthresh = host.get_subset_proportion()[s];
376 for (I<decltype(num_cells)> i = 0; i < num_cells; ++i) {
377 const auto thresh = [&]{
378 if constexpr(unblocked) {
379 return sthresh;
380 } else {
381 return sthresh[block[i]];
382 }
383 }();
384 output[i] = output[i] && (sub[i] <= thresh);
385 }
386 }
387}
