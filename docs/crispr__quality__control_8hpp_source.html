228template<typename Float_, class Host_, typename Sum_, typename Detected_, typename Value_, typename Index_, typename BlockSource_>
229void crispr_populate(
230 Host_& host,
231 const std::size_t num_cells,
232 const ComputeCrisprQcMetricsBuffers<Sum_, Detected_, Value_, Index_>& res,
233 BlockSource_ block,
234 const std::size_t num_blocks,
235 const ComputeCrisprQcFiltersOptions& options
236) {
237 constexpr bool unblocked = std::is_same<BlockSource_, bool>::value;
238 auto buffer = [&]{
239 if constexpr(unblocked) {
240 return sanisizer::create<std::vector<Float_> >(num_cells);
241 } else {
242 return FindMedianMadBlockedWorkspace<Float_>(num_cells, block, num_blocks);
243 }
244 }();
245
246 // Subsetting to the observations in the top 50% of proportions.
247 static_assert(std::is_floating_point<Float_>::value);
248 std::vector<Float_> maxprop;
249 maxprop.reserve(num_cells);
250 for (I<decltype(num_cells)> i = 0; i < num_cells; ++i) {
251 maxprop.push_back(static_cast<Float_>(res.max_value[i]) / static_cast<Float_>(res.sum[i]));
252 }
253
254 FindMedianMadOptions fopt;
255 fopt.median_only = true;
256 auto prop_res = [&]{
257 if constexpr(unblocked) {
258 std::copy_n(maxprop.begin(), num_cells, buffer.begin());
259 return find_median_mad(num_cells, buffer.data(), fopt);
260 } else {
261 return find_median_mad_blocked(num_cells, maxprop.data(), block, num_blocks, &buffer, fopt);
262 }
263 }();
264
265 for (I<decltype(num_cells)> i = 0; i < num_cells; ++i) {
266 auto limit = [&]{
267 if constexpr(unblocked){
268 return prop_res.median;
269 } else {
270 return prop_res[block[i]].median;
271 }
272 }();
273 if (maxprop[i] >= limit) {
274 maxprop[i] = res.max_value[i];
275 } else {
276 maxprop[i] = std::numeric_limits<Float_>::quiet_NaN(); // ignored during threshold calculation.
277 }
278 }
279
280 // Filtering on the max counts.
281 ChooseFilterThresholdsOptions copt;
282 copt.num_mads = options.max_value_num_mads;
283 copt.log = true;
284 copt.upper = false;
285 host.get_max_value() = [&]{
286 if constexpr(unblocked) {
287 std::copy_n(maxprop.begin(), num_cells, buffer.begin());
288 return choose_filter_thresholds(num_cells, buffer.data(), copt).lower;
289 } else {
290 return internal::strip_threshold<true>(choose_filter_thresholds_blocked(num_cells, maxprop.data(), block, num_blocks, &buffer, copt));
291 }
292 }();
293}

295template<class Host_, typename Sum_, typename Detected_, typename Value_, typename Index_, typename BlockSource_, typename Output_>
296void crispr_filter(const Host_& host, const std::size_t n, const ComputeCrisprQcMetricsBuffers<Sum_, Detected_, Value_, Index_>& metrics, BlockSource_ block, Output_* const output) {
297 constexpr bool unblocked = std::is_same<BlockSource_, bool>::value;
298 std::fill_n(output, n, 1);
299
300 const auto& mv = host.get_max_value();
301 for (I<decltype(n)> i = 0; i < n; ++i) {
302 auto thresh = [&]{
303 if constexpr(unblocked) {
304 return mv;
305 } else {
306 return mv[block[i]];
307 }
308 }();
309 output[i] = output[i] && (metrics.max_value[i] >= thresh);
310 }
311}
